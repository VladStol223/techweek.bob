**Coordinate conversion:** JSON stores positions as pixel x/y in 1792×1536 space:
  lon_left: -88.59375, lon_right: -78.75, lat_top: 32.99023, lat_bottom: 24.20688
```js
function xyToLatLng(x, y) {
  const lng = -88.59375 + (x / 1792) * (-78.75 - -88.59375);
  const lat =  32.99023 - (y / 1536) * ( 32.99023 - 24.20688);
  return [lat, lng];
}
```

**Truck icon:** L.divIcon — a small colored rectangle (12×18px) with a rounded top (cab), rotated to face direction of travel. Colors: green (#22c55e) for on_time, yellow (#eab308) for delayed, grey (#6b7280) for returned/delivered.

**Animation loop** — setInterval at 100ms. For each delivery in WD.deliveries:
```js
const CYCLE_MS = 120000;
function getTruckState(delivery, index) {
  const phase = ((Date.now() + (index / WD.deliveries.length) * CYCLE_MS) % CYCLE_MS) / CYCLE_MS;
  const wh = WD.warehouses.find(w => w.id === delivery.from_warehouse_id);
  const st = WD.stores.find(s => s.id === delivery.to_store_id);
  const [whLat, whLng] = window.xyToLatLng(wh.x, wh.y);
  const [stLat, stLng] = window.xyToLatLng(st.x, st.y);
  let lat, lng, dLat, dLng;
  if (phase < 0.5) {
    const t = phase / 0.5;
    lat = whLat + (stLat - whLat) * t; lng = whLng + (stLng - whLng) * t;
    dLat = stLat - whLat; dLng = stLng - whLng;
  } else {
    const t = (phase - 0.5) / 0.5;
    lat = stLat + (whLat - stLat) * t; lng = stLng + (whLng - stLng) * t;
    dLat = whLat - stLat; dLng = whLng - stLng;
  }
  const angleDeg = Math.atan2(dLng, dLat) * 180 / Math.PI;
  const liveStatus = phase < 0.5 ? delivery.status : 'delivered';
  const color = liveStatus === 'delayed' ? '#eab308' : liveStatus === 'delivered' ? '#6b7280' : '#22c55e';
  return { lat, lng, angleDeg, direction: phase < 0.5 ? 'Outbound' : 'Returning', liveStatus, color };
}
```
